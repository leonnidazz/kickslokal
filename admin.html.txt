    <script>
        // MASUKKAN URL & PUBLISHABLE KEY SUPABASE KAMU DI SINI
        const SUPABASE_URL = 'GANTI_DENGAN_PROJECT_URL_KAMU';
        const SUPABASE_ANON_KEY = 'GANTI_DENGAN_PUBLISHABLE_KEY_KAMU';

        const { createClient } = supabase;
        const _supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

        async function addNewProduct(e) {
            e.preventDefault();
            const btn = document.getElementById('submit-btn');
            btn.innerText = 'Menyimpan...';
            btn.disabled = true;

            const name = document.getElementById('p-name').value;
            const brand = document.getElementById('p-brand').value;
            const price = parseInt(document.getElementById('p-price').value);
            const type = document.getElementById('p-type').value;
            const pengguna = document.getElementById('p-pengguna').value;
            const sizes = document.getElementById('p-sizes').value;
            const image_url = document.getElementById('p-image').value;

            const { data, error } = await _supabase.from('products').insert([
                { name, brand, price, type, pengguna, sizes, image_url }
            ]);

            if (error) {
                alert('Gagal menyimpan produk: ' + error.message);
                btn.innerText = 'Simpan Produk ke Database';
                btn.disabled = false;
            } else {
                alert('Berhasil! Produk baru telah masuk ke database.');
                document.getElementById('product-form').reset();
                btn.innerText = 'Simpan Produk ke Database';
                btn.disabled = false;
            }
        }
    </script>
